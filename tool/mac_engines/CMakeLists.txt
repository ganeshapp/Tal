cmake_minimum_required(VERSION 3.14)
project(still_stockfish LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(SF_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../../third_party/stockfish)
file(GLOB_RECURSE SF_SOURCES ${SF_DIR}/ios/Stockfish/src/*.cpp)
add_executable(tal-stockfish ${SF_SOURCES} streams.cpp)
target_compile_definitions(tal-stockfish PRIVATE USE_PTHREADS IS_64BIT USE_POPCNT USE_NEON=8)
target_compile_options(tal-stockfish PRIVATE -O3 -DNDEBUG -include ${SF_DIR}/ios/FlutterStockfish/still_streams.h)
file(GLOB NETWORKS ${SF_DIR}/networks/*.nnue)
foreach(NETWORK ${NETWORKS})
 get_filename_component(NET_NAME ${NETWORK} NAME)
 configure_file(${NETWORK} ${CMAKE_CURRENT_BINARY_DIR}/${NET_NAME} COPYONLY)
endforeach()
